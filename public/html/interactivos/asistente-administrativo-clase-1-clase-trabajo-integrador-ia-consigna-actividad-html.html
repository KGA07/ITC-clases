BASE DE DATOS + GOOGLE APPS SCRIPT

⏱ 40 minutos

Crea tu propia base de datos backend usando Google Sheets y Google Apps Script. Deberás desarrollar un sistema completo con código .gs y .html dentro del editor de Apps Script.

  2. Nómbrala: "BD_[TuNombre]_[Tema]" (ej: BD_JuanPerez_ControlDocumentos)

Paso 2: Abrir el editor de Apps Script

  1. En tu hoja de cálculo, ve a Extensiones → Apps Script

// ===== CONFIGURACIÓN =====
var HOJA_NAME = "Datos";
var COL_NOMBRE = 1;
var COL_EMAIL = 2;
var COL_FECHA_INGRESO = 3;
var COL_FECHA_VENCIMIENTO = 4;
var COL_ESTADO = 5;
var COL_ENVIADO = 6;

// ===== FUNCIÓN PRINCIPAL: Verificar vencimientos =====
function verificarVencimientos() {
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_NAME);
  var datos = hoja.getDataRange().getValues();

  for (var i = 1; i < datos.length; i++) {
    var fila = i + 1;
    var fechaVenc = new Date(datos[i][COL_FECHA_VENCIMIENTO - 1]);
    var estado = calcularEstado(fechaVenc);
    var yaEnviado = datos[i][COL_ENVIADO - 1];

    hoja.getRange(fila, COL_ESTADO).setValue(estado);

    if (estado === "VENCIDO" && yaEnviado !== "Sí") {
      var nombre = datos[i][COL_NOMBRE - 1];
      var email = datos[i][COL_EMAIL - 1];
      enviarNotificacion(nombre, email, fechaVenc);
      hoja.getRange(fila, COL_ENVIADO).setValue("Sí");
    }
  }
}

// ===== CALCULAR ESTADO SEGÚN FECHA =====
function calcularEstado(fechaVenc) {
  var hoy = new Date();
  var diffDias = Math.floor((fechaVenc - hoy) / (1000 * 60 * 60 * 24));

  if (diffDias < 0) return "VENCIDO";
  if (diffDias <= 7) return "POR VENCER";
  return "VIGENTE";
}

// ===== ENVIAR NOTIFICACIÓN POR EMAIL =====
function enviarNotificacion(nombre, email, fechaVenc) {
  var asunto = "ALUMNO IA ITC 2026 - Documento Vencido";
  var mensaje = "Estimado/a " + nombre + ",\n\n" +
                "Su registro ha alcanzado estado VENCIDO.\n" +
                "Fecha de vencimiento: " + fechaVenc.toLocaleDateString() + "\n\n" +
                "Por favor regularice su situación.\n\n" +
                "Atentamente,\nSistema Automatizado ITC 2026";

  try {
    MailApp.sendEmail(email, asunto, mensaje);
    Logger.log("Notificación enviada a: " + email);
  } catch (e) {
    Logger.log("Error al enviar a " + email + ": " + e.message);
  }
}

// ===== FUNCIONES PARA LA APP (API) =====
function doGet(e) {
  var accion = e.parameter.accion;

  if (accion === "obtener") {
    return ContentService.createTextOutput(
      JSON.stringify(obtenerDatos())
    ).setMimeType(ContentService.MimeType.JSON);
  }

  if (accion === "agregar") {
    var resultado = agregarDato(
      e.parameter.nombre,
      e.parameter.email,
      e.parameter.fechaIngreso,
      e.parameter.fechaVencimiento
    );
    return ContentService.createTextOutput(
      JSON.stringify(resultado)
    ).setMimeType(ContentService.MimeType.JSON);
  }
}

function obtenerDatos() {
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_NAME);
  var datos = hoja.getDataRange().getValues();
  var resultado = [];

  for (var i = 1; i < datos.length; i++) {
    resultado.push({
      nombre: datos[i][COL_NOMBRE - 1],
      email: datos[i][COL_EMAIL - 1],
      fechaIngreso: datos[i][COL_FECHA_INGRESO - 1],
      fechaVencimiento: datos[i][COL_FECHA_VENCIMIENTO - 1],
      estado: datos[i][COL_ESTADO - 1]
    });
  }

  return resultado;
}

function agregarDato(nombre, email, fechaIngreso, fechaVencimiento) {
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_NAME);
  var estado = calcularEstado(new Date(fechaVencimiento));

  hoja.appendRow([nombre, email, fechaIngreso, fechaVencimiento, estado, "No"]);

  return { "status": "ok", "mensaje": "Dato registrado exitosamente" };
}

// ===== CREAR MENÚ PERSONALIZADO =====
function onOpen() {
  var ui = SpreadsheetApp.getUi();
  ui.createMenu('🤖 Sistema ITC')
    .addItem('Verificar Vencimientos', 'verificarVencimientos')
    .addItem('Actualizar Estados', 'actualizarEstados')
    .addToUi();
}

function actualizarEstados() {
  var hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(HOJA_NAME);
  var datos = hoja.getDataRange().getValues();

  for (var i = 1; i < datos.length; i++) {
    var fila = i + 1;
    var fechaVenc = new Date(datos[i][COL_FECHA_VENCIMIENTO - 1]);
    var estado = calcularEstado(fechaVenc);
    hoja.getRange(fila, COL_ESTADO).setValue(estado);
  }

  SpreadsheetApp.getUi().alert('Estados actualizados correctamente');
}

    function agregarDato() {
      var nombre = document.getElementById('nombre').value;
      var email = document.getElementById('email').value;
      var fechaIngreso = document.getElementById('fechaIngreso').value;
      var fechaVencimiento = document.getElementById('fechaVencimiento').value;

      google.script.run
        .withSuccessHandler(function(resp) {
          document.getElementById('resultado').style.display = 'block';
          document.getElementById('resultado').textContent = '✅ ' + resp.mensaje;
          cargarDatos();
        })
        .agregarDato(nombre, email, fechaIngreso, fechaVencimiento);
    }

    function cargarDatos() {
      google.script.run
        .withSuccessHandler(function(datos) {
          var html = '';
          datos.forEach(function(d) {
            var clase = 'estado-' + d.estado.replace(/\s/g, '-');
            html += '<div class="dato-item ' + clase + '">';
            html += '<strong>' + d.nombre + '</strong><br>';

            html += '📅 Vence: ' + d.fechaVencimiento + '<br>';
            html += '📌 Estado: <strong>' + d.estado + '</strong>';
            html += '</div>';
          });
          document.getElementById('lista').innerHTML = html;
        })
        .obtenerDatos();
    }

    cargarDatos();
  </script>
